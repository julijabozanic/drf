# 1. CSRF
python -m httpie --session=s GET http://127.0.0.1:8000/api/auth/csrf/

# 2. Register
python -m httpie --session=s POST http://127.0.0.1:8000/api/auth/register/ username=petar email=petar@gmail.com password=SuperJak123 password2=SuperJak123 X-CSRFToken:<token_from_step_1>

# 3. Reset Axes before testing lockout (clean start)
uv run manage.py axes_reset

# 4. Lockout — AXES_FAILURE_LIMIT is 10 in .env, perform 10 failed attempts (cmd)
for /L %i in (1,1,3) do python -m httpie --session=s POST http://127.0.0.1:8000/api/auth/login/ username=petar password=wrong X-CSRFToken:wFGZWfOatyH6T4pna9Txn83eK6r2yICl
# Expected: the 3rd attempt (or earlier, depending on AXES_LOCKOUT_PARAMETERS) -> 403 from AxesMiddleware, not 400 from the serializer

# 5. Reset after the test
uv run manage.py axes_reset

# 6. Fresh CSRF token — the old one is invalid after login()/register()
python -m httpie --session=s GET http://127.0.0.1:8000/api/auth/csrf/

# 7. Login with valid credentials
python -m httpie --session=s POST http://127.0.0.1:8000/api/auth/login/ username=petar password=SuperJak123 X-CSRFToken:<new_token>

# 8. Create an issue (as Petar)
python -m httpie --session=s POST http://127.0.0.1:8000/api/issues/ title="Test" description="Description" X-CSRFToken:<new_token>

# 9. Attempt to change the status as a regular user — should be silently ignored (status remains 'open')
python -m httpie --session=s PATCH http://127.0.0.1:8000/api/issues/1/ status=closed X-CSRFToken:<new_token>

# 10. Comment
python -m httpie --session=s POST http://127.0.0.1:8000/api/issues/1/comments/ body="Test comment" X-CSRFToken:<new_token>

# ---- Authorization with two accounts ----

# 11. Register a second user (new session!)
python -m httpie --session=userB GET http://127.0.0.1:8000/api/auth/csrf/
python -m httpie --session=userB POST http://127.0.0.1:8000/api/auth/register/ username=marko email=marko@gmail.com password=DrugiPass123 password2=DrugiPass123 X-CSRFToken:<token_userB>

# 12. User B attempts to comment on Petar's issue (id=1) — expect 404, not 403/200
python -m httpie --session=userB GET http://127.0.0.1:8000/api/auth/csrf/
python -m httpie --session=userB POST http://127.0.0.1:8000/api/issues/1/comments/ body="Someone else's comment" X-CSRFToken:<fresh_token_userB>

# ---- Admin verification ----

# 13. Manually change role=admin in the Django admin panel (/admin/) or use a createsuperuser account
# 14. Login as admin (new session)
python -m httpie --session=admin GET http://127.0.0.1:8000/api/auth/csrf/
python -m httpie --session=admin POST http://127.0.0.1:8000/api/auth/login/ username=<admin_username> password=<admin_pass> X-CSRFToken:<token_admin>

# 15. Admin can see ALL issues (including Petar's, not only their own)
python -m httpie --session=admin GET http://127.0.0.1:8000/api/issues/

# 16. Admin can successfully change the status (unlike step 9)
python -m httpie --session=admin GET http://127.0.0.1:8000/api/auth/csrf/
python -m httpie --session=admin PATCH http://127.0.0.1:8000/api/issues/1/ status=closed X-CSRFToken:<fresh_token_admin>
